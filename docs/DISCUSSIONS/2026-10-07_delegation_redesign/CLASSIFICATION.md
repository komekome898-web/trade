### 数えた台本 `count.pl` の本文

```perl
use utf8; use strict; use warnings;
binmode STDOUT, ':encoding(UTF-8)';
my $mode = shift @ARGV;
my (%t, %k, %kt, %sym, $torn, $nodel, $rows);
my @order = ('a','b','c','d','e','f','g');
while (my $l = <>) {
  utf8::decode($l);
  next unless $l =~ /^\| \d+ \|/;
  my @f = split /\|/, $l;
  (my $t = $f[5]) =~ s/\s//g;
  my ($key) = $f[4] =~ /(〔[^〕]*〕)/;
  my $is_y = ($f[2] =~ /\[直すとよい/) ? 1 : 0;
  next if $mode eq 'noY' && $is_y;
  $rows++;
  $t{$t}++; $k{$key}++; $kt{$key}{$t}++;
  $torn++ if $t =~ m{/};
  $nodel++ if $t eq 'g' && $f[6] =~ /^\s*委任文なし。/;
  $sym{$_}++ for split m{/}, $t;
}
if ($mode eq 'type' || $mode eq 'noY') {
  print "| 原因の型(型の列そのまま) | 件数 |\n|---|---|\n";
  print "| $_ | $t{$_} |\n" for sort keys %t;
  print "| 計 | $rows |\n\n";
  print "- 迷った行(型の列に / がある): $torn\n";
  print "- (g) のうち根拠が「委任文なし。」で始まる行(関門 ② の読み): $nodel\n";
  print "- 型の記号ごとの出現(迷った行は両方の記号に 1 ずつ数える): ", join('、', map { "$_ " . ($sym{$_} // 0) } @order), "\n";
}
if ($mode eq 'deleg') {
  print "| 委任の鍵 | ", join(' | ', map { "($_)" } @order), " | 迷った行 | 計 |\n";
  print "|---|", join('', map { '---|' } @order), "---|---|\n";
  for my $key (sort keys %k) {
    my %s; my $tn = 0;
    for my $t (keys %{$kt{$key}}) { if ($t =~ m{/}) { $tn += $kt{$key}{$t}; } else { $s{$t} += $kt{$key}{$t}; } }
    my @tl = map { "$_ $kt{$key}{$_}" } grep { m{/} } sort keys %{$kt{$key}};
    print "| $key | ", join(' | ', map { $s{$_} // 0 } @order), " | ", ($tn ? "$tn(" . join('、', @tl) . ")" : 0), " | $k{$key} |\n";
  }
}
```
